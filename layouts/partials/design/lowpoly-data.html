{{- /* Inline every background's low-poly SVG as a URL-encoded data-URI so the randomiser (design.js) can show an instant faceted placeholder for whichever photo it picks — with no extra network request. Emitted only on marketing pages (see extend-footer.html), before the deferred design.js runs, so window.CS_LOWPOLY is set by the time the randomiser executes. URL-encoding (rather than base64) skips the ~33% size overhead and compresses better under gzip — see design/svg-url-encode.html. */ -}} {{- $uris := slice -}} {{- range $i := (seq 1 10) -}} {{- $n := printf "%02d" $i -}} {{- $raw := os.ReadFile (printf "static/images/backgrounds/lowpoly/%s.svg" $n) -}} {{- $enc := partial "design/svg-url-encode.html" $raw -}} {{- $uris = $uris | append (printf "data:image/svg+xml,%s" $enc) -}} {{- end -}}
