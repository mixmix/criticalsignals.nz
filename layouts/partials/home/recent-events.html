{{- /* Home — Recent Events: a random sample of photos from the last 5 past events that actually have gallery photos (each event's "extra" photos — see programme/gallery-images.html — not its header image). Events without any gallery photos are skipped over, scanning further back in time as needed, so a run of photo-less events (e.g. recurring drop-ins) can't leave the gallery empty. Every candidate photo is rendered into a JSON data island on the grid element; design.js shuffles it and builds the N visible thumbnails client-side, so the sample is different on every page load. N is read from the front matter (`recentEventsCount`), defaulting to 6. Renders nothing if there are no candidate photos at all. Thumbnails are converted to actual greyscale pixels at build time (images.Grayscale), matching partials/programme/gallery.html. */ -}} {{- $n := .Params.recentEventsCount | default 6 -}} {{- $today := (partial "now-nzt.html" .).Format "2006-01-02" -}} {{- $events := where .Site.RegularPages "Section" "programme" -}} {{/* One entry per past event, keyed by page so a recurring event (multiple occurrences) is only counted once, at its most recent occurrence. */}} {{- $byPage := dict -}} {{- range partial "programme/occurrences.html" $events -}} {{- if and (not .page.Draft) (lt .day $today) -}} {{- $key := .page.RelPermalink -}} {{- $existing := index $byPage $key -}} {{- if or (not $existing) (gt .day $existing.day) -}} {{- $byPage = merge $byPage (dict $key .) -}} {{- end -}} {{- end -}} {{- end -}} {{- $allRecent := slice -}} {{- range $byPage -}}{{- $allRecent = $allRecent | append . -}}{{- end -}} {{- $allRecent = sort $allRecent "day" "desc" -}} {{/* Walk back through past events, skipping any with no gallery photos, until 5 photo-bearing events are found (or the history runs out). */}} {{- $withPhotos := slice -}} {{- range $allRecent -}} {{- if lt (len $withPhotos) 5 -}} {{- $p := .page -}} {{- $imgs := partial "programme/gallery-images.html" $p -}} {{- if gt (len $imgs) 0 -}} {{- $withPhotos = $withPhotos | append (dict "page" $p "images" $imgs) -}} {{- end -}} {{- end -}} {{- end -}} {{- $pool := slice -}} {{- range $withPhotos -}} {{- $p := .page -}} {{- range .images -}} {{- $thumb := (.Resize (printf "%dx webp q80" (cond (gt .Width 300) 300 .Width))).Filter images.Grayscale -}} {{- $full := .Resize (printf "%dx webp q90" .Width) -}} {{- $pool = $pool | append (dict "src" $full.RelPermalink "thumb" $thumb.RelPermalink "width" $thumb.Width "height" $thumb.Height "fullWidth" $full.Width "fullHeight" $full.Height "title" $p.Title "href" $p.RelPermalink) -}} {{- end -}} {{- end -}} {{- with $pool -}} {{- /* However many slots will actually be filled — the pool can be smaller than the requested count. Rendered as empty `.gallery-item`s so the grid occupies its final size in the HTML itself, before design.js has run: each one is the same fixed aspect-ratio box, with the same grey noise backdrop, that a filled item occupies (see custom.css), so the page doesn't change shape when the thumbnails are swapped in. */ -}} {{- $slots := cond (lt (len .) $n) (len .) $n -}}
